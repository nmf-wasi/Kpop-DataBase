from app.database.database import Base
from sqlalchemy.orm import mapped_column, Mapped, relationship
from sqlalchemy import Integer, String, DateTime
from app.config.enums import UserRole
from datetime import datetime
from datetime import date


class User(Base):
    __tablename__ = "users"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)

    first_name: Mapped[str | None] = mapped_column(String, nullable=True)
    last_name: Mapped[str | None] = mapped_column(String, nullable=True)

    username: Mapped[str] = mapped_column(String, unique=True)
    email: Mapped[str | None] = mapped_column(String, unique=True)
    hashed_password: Mapped[str] = mapped_column(String)

    role: Mapped[UserRole] = mapped_column(default=UserRole.USER)

    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.now)
    # dont call datetime.now() or else it will be executed everytime we touch a row
    updated_profile_at: Mapped[datetime] = mapped_column(
        DateTime, default=datetime.now, onupdate=datetime.now
    )
    last_login: Mapped[datetime | None] = mapped_column(
        DateTime, nullable=True, default=None
    )


class Song(Base):
    """Song table"""

    __tablename__ = "songs"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    # Mapped[X]
    # This is just a type annotation — it tells Python what Python type this attribute will be when you access it on an instanc

    # mapped_column(...)
    # This is the actual column definition — the thing that becomes a real column in the database table.
    # It's the 2.0-style replacement for the old Column(...).

    title: Mapped[str] = mapped_column(String)
    release_date: Mapped[date] = mapped_column(String)
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.now)
    last_update: Mapped[datetime] = mapped_column(DateTime, default=datetime.now)
    album_id: Mapped[int] = mapped_column(Integer, ForeignKey=("albums.id"))
    # The rule: the FK always lives on the "many" side
    # Album ↔ Song​  -> Song gets the foreign key, cause many songs, 1 album

    # Group ↔ Album​ -> Album gets the foreign key
    # album_id is the actual foreign key stored in the songs table

    album: Mapped["Album"] = relationship(back_populates="songs")
    # back populates song col on Album table

    # relationship(...)

    # This is not a column. No new field appears in the database table. It's a Python-only construct that tells the ORM: "when someone accesses song.album, go fetch the related Album row for me by following the FK."

    # back_populates

    # This is what links two relationship() calls into one bidirectional relationship, instead of two independent, unaware-of-each-other relationships.

    # The FK column and the relationship() are two separate, independent things.
    # You could have album_id with no relationship() at all — perfectly valid, just less convenient. Then song.album_id gives you 5, and if you want the actual Album row, you'd manually do session.get(Album, song.album_id) every time.
    # relationship() just saves you that manual lookup
